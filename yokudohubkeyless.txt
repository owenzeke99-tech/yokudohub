-- Tên file lưu trữ cache (Lưu trạng thái đã nhập key đúng trong 24h)
local saveFileName = "YokudoHub_KeyCache.json"
local keyDuration = 86400 -- 24 giờ tính bằng giây (24 * 60 * 60)

-- Hàm kiểm tra xem đã lưu key và còn hạn hay không
local function checkSavedKey()
    if readfile and writefile and isfile and isfile(saveFileName) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(saveFileName))
        end)
        if success and data and data.timestamp then
            if os.time() - data.timestamp < keyDuration then
                return true -- Vẫn còn hạn 24 giờ
            end
        end
    end
    return false
end

-- Hàm lưu thời gian khi nhập key đúng
local function saveKeyData()
    if writefile then
        local data = { timestamp = os.time() }
        pcall(function()
            writefile(saveFileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Nếu đã lưu key hợp lệ trong vòng 24h thì bỏ qua bảng key và load thẳng script chính
if checkSavedKey() then
    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/betdoyvaka/stealanegg/main/Loader.lua"))()
    end)
    if not success then
        warn("Error loading main script: " .. tostring(err))
    end
    return -- Dừng chạy code bên dưới để không hiện bảng UI nữa
end

local CORRECT_KEY = "YokudoHubKeyGdjsi - Nsksowiwjwooenekeek"

-- Hàm hiệu ứng lắc kết hợp xóa chữ
local function shakeAndClear()
    KeyBox.Text = ""
    local originalPos = UDim2.new(0.5, -210, 0.5, -120)
    local intensity = 8
    
    for i = 1, 3 do
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, intensity, 0, 0)}):Play()
        task.wait(0.04)
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, -intensity, 0, 0)}):Play()
        task.wait(0.04)
    end
    TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos}):Play()
end

-- Xử lý logic Xác Nhận Key
SubmitBtn.MouseButton1Click:Connect(function()
    if KeyBox.Text == CORRECT_KEY then
        StatusLabel.TextColor3 = Color3.fromRGB(90, 255, 130)
        StatusLabel.Text = "✔ Key correct! Loading script..."
        
        -- Lưu cache 24h
        saveKeyData()
        
        task.wait(1)
        
        -- Xóa Menu Key
        ScreenGui:Destroy()
        
        -- Load Script chính
        local success, err = pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/betdoyvaka/stealanegg/main/Loader.lua"))()
        end)
        
        if not success then
            warn("Error loading main script: " .. tostring(err))
        end
    else
        StatusLabel.TextColor3 = Color3.fromRGB(255, 90, 90)
        StatusLabel.Text = "✖ Invalid key! Please try again."
        shakeAndClear()
    end
end)
